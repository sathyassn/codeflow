## Repository, release, security, and cross-build commands

~~~sh
codeflow test --strict

cargo dist generate-ci --check
cargo dist plan --output-format=json > /tmp/codeflow-v3-dist-plan.json
jq -e '
  .announcement_tag == "v3.0.0"
  and (.releases[0].artifacts | index("codeflow-cli-aarch64-apple-darwin.tar.xz") != null)
  and (.releases[0].artifacts | index("codeflow-cli-x86_64-apple-darwin.tar.xz") != null)
  and (.releases[0].artifacts | index("codeflow-cli-x86_64-unknown-linux-gnu.tar.xz") != null)
  and (.releases[0].artifacts | index("codeflow-cli-x86_64-pc-windows-msvc.zip") != null)
  and (.releases[0].artifacts | index("codeflow-cli-installer.sh") != null)
  and (.releases[0].artifacts | index("codeflow-cli-installer.ps1") != null)
' /tmp/codeflow-v3-dist-plan.json > /dev/null
/usr/bin/trash /tmp/codeflow-v3-dist-plan.json

cargo audit
cargo xwin check --workspace --target x86_64-pc-windows-msvc
cargo xwin build --workspace --release --target x86_64-pc-windows-msvc
cargo zigbuild --workspace --release --target x86_64-unknown-linux-gnu
cargo build --workspace --release
test "$(target/release/codeflow --version)" = "codeflow 3.0.0"
test "$(git cliff --bumped-version)" = "v3.0.0"
gitleaks git . --config .gitleaks.toml --log-opts="--all" \
  --redact --no-banner
~~~

## Brownfield canary

~~~sh
set -e
newbin="$PWD/target/release/codeflow"
oldbin=$(command -v codeflow)
test "$("$newbin" --version)" = "codeflow 3.0.0"
test "$("$oldbin" --version)" = "codeflow 2.1.0"
brown=$(mktemp -d /tmp/codeflow-v3-brownfield.XXXXXX)
cd "$brown"
git init -q
git config user.email canary@example.invalid
git config user.name "CodeFlow Canary"
"$oldbin" init --full --yes > /tmp/codeflow-v3-brown-init.log
git switch -qc test/brownfield
cp docs/product.md README.md
perl -0pi -e 's/\z/\nBrownfield owner README marker.\n/' README.md
perl -0pi -e 's/\z/\nBrownfield owner product marker.\n/' docs/product.md
perl -0pi -e 's/\z/\nBrownfield owner agent instruction.\n/' AGENTS.md
git add -A
git commit -qm "test: establish v2 brownfield"
readme_before=$(shasum -a 256 README.md | awk '{print $1}')
product_before=$(shasum -a 256 docs/product.md | awk '{print $1}')
"$newbin" update > /tmp/codeflow-v3-brown-update-1.log
"$newbin" update > /tmp/codeflow-v3-brown-update-2.log
rg -q "107 unchanged" /tmp/codeflow-v3-brown-update-2.log
test "$readme_before" = "$(shasum -a 256 README.md | awk '{print $1}')"
test "$product_before" = "$(shasum -a 256 docs/product.md | awk '{print $1}')"
rg -q "Brownfield owner agent instruction\." AGENTS.md
"$newbin" validate --docs
"$newbin" doctor --check repo-integrity
"$newbin" doctor --check managed-drift
"$newbin" doctor --check test-config
test ! -e docs-portal
test ! -e .codeflow-present
test ! -e .codeflow/present
cd /
/usr/bin/trash "$brown"
/usr/bin/trash \
  /tmp/codeflow-v3-brown-init.log \
  /tmp/codeflow-v3-brown-update-1.log \
  /tmp/codeflow-v3-brown-update-2.log
~~~

Both canaries passed. Repeated greenfield init was idempotent. The brownfield
preserved exact owner README/product hashes and its owner instruction outside
the managed AGENTS block; its second v3 update reported 107 unchanged managed
files. Neither canary installed an opt-in presentation or portal runtime.
